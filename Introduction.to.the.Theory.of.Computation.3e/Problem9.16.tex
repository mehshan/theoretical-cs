\begin{problem}{9.16}
Prove that $TQBF \notin \text{SPACE}(n^{1/3})$.
\end{problem}

\begin{proof}
We know that
\[
TQBF \in \text{PSPACE},
\]
and
\[
\text{SPACE}(n) \subseteq \text{PSPACE}.
\]
According to Space hierarchy theorem
\[
\text{SPACE}(n^{1/3}) \subsetneq \text{SPACE}(n).
\]

Therefore,
\[
TQBF \notin \text{SPACE}(n^{1/3}).
\]
\end{proof}
