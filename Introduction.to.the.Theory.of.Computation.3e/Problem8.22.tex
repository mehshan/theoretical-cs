\begin{problem}[Part]{a}
Let $ADD = \{\langle x, y, z \rangle \ | \ x, y, z > 0 \text{ are binary integers and } x + y = z\}$. \\ Show that $ADD \in$ L.
\end{problem}

\begin{proof}
To show that $ADD \in$ L, we give a log space algorithm $M$ that decides $ADD$. \\

$M =$ \textquotedblleft On input $\langle x, y, z \rangle$, where $x$, $y$ and $z$ are binary integers greater than 0:
\begin{enumerate}
\item Let $i$, $n_x$, $n_y$, $n_z$ and $m$ be binary integers, and $s$ and $c$ be binary digits on work tape.
\item Let $n_x$ = $|x|$, $n_y$ = $|y|$, $n_z = |z|$, and $m$ be maximum of $n_x$ and $n_y$.
\item Let $i = 1$, and $c$ = 0.
\item Repeat for $m$ steps:
\item \hspace*{0.5cm} If $z$ does not have $i^{th}$ digit $(i > n_z)$, then \textit{reject}.
\item \hspace*{0.5cm} Let $s$ be the sum of $c$ and $i^{th}$ binary digits of $x$ and $y$. If either $x$ or $y$ does not have $i^{th}$ \\ \hspace*{0.5cm} binary digit, then assume the digit to be 0. 
\item \hspace*{0.5cm} If the addition produces a carry, then $c = 1$. Otherwise, $c = 0$.
\item \hspace*{0.5cm} If $s$ and $i^{th}$ digit of $z$ are different, then \textit{reject}.
\item \hspace*{0.5cm} Increment $i$ by 1.
\item If $c = 1$, then:
\item \hspace*{0.5cm} If $z$ does not have $i^{th}$ digit, then \textit{reject}.
\item \hspace*{0.5cm} If $c$ and $i^{th}$ digit of $z$ are different, then \textit{reject}.
\item \hspace*{0.5cm} If $z$ has $(i+1)^{th}$ digit, then \textit{reject}.
\item Otherwise, if $z$ has $i^{th}$ digit, then \textit{reject}. 
\item \textit{Accept}.\textquotedblright
\end{enumerate}
\end{proof}

\begin{problem}[Part]{b}
Let $PAL-ADD = \{\langle x, y \rangle \ | \ x, y > 0 \text{ are binary integers where } x + y \text{ is an integer whose } \\ \text{binary representation is a palindrome}\}$. (Note that the binary
representation of the sum is assumed not to have leading zeros. A palindrome is a string that equals its reverse.) \\ Show that $PAL-ADD \in$ L.
\end{problem}

\begin{proof} 
To show that $PAL-ADD \in$ L, we give a log space algorithm $M$ that decides $PAL-ADD$. \\

$M =$ \textquotedblleft On input $\langle x, y \rangle$, where $x$ and $y$ are binary integers greater than 0:
\begin{enumerate}
\item Let $i$, $j$ and $m$ be binary integers on work tape.
\item Let $m$ be the maximum of $|x|$ and $|y|$.
\item Run $C$ on input $\langle x, y \rangle$. If $C$ accepts then $m = m + 1$.
\item Let $i = 1$, and $j = m$.
\item Repeat for $m - 1$ steps:
\item \hspace*{0.5cm} Run $B$ on $\langle x, y, i\rangle$. If $B$ accepts, then $b_i = 1$. Otherwise $b_i$ = 0.
\item \hspace*{0.5cm} Run $B$ on $\langle x, y, j\rangle$. If $B$ accepts, then $b_j = 1$. Otherwise $b_j$ = 0.
\item \hspace*{0.5cm} If $b_i$ and $b_j$ are different, then \textit{reject}.
\item \hspace*{0.5cm} Increment $i$ by 1, and decrement $j$ by 1.
\item \textit{Accept}.\textquotedblright
\end{enumerate}

Let $CARRY = \{\langle x, y \rangle \ | \ x, y > 0 \text{ are binary integers whose sum produces a carry}\}$. The algorithm $C$ decides $CARRY$ in log space. \\

$C =$ \textquotedblleft On input $\langle x, y \rangle$, where $x$ and $y$ are binary integers greater than 0:
\begin{enumerate}
\item Let $i$ and $m$ be binary integers on work tape where $i = 1$ and $m = max(n_x$, $n_y)$. 
\item Let $s$ and $c$ be binary digits on work tape where $s, \ c = 0$.
\item Repeat for $m$ steps:
\item \hspace*{0.5cm} Let $s$ be the sum of $c$ and $i^{th}$ binary digits of $x$ and $y$. If either $x$ or $y$ does not have \\
\hspace*{0.5cm} $i^{th}$ binary digit, then the digit is 0.
\item \hspace*{0.5cm} If the addition produces a carry, then $c = 1$. 
Otherwise, $c = 0$.
\item \hspace*{0.5cm} Increment $i$ by 1.
\item If $c = 1$ then \textit{accept}. Otherwise, \textit{reject}.\textquotedblright
\end{enumerate}

Let $BIT = \{\langle x, y, i \rangle \ | \ x, y, i > 0 \text{ are binary integers, and } i^{th} \text{ digit of } x + y \text{ is } 1\}$. The algorithm $B$ decides $BIT$ in log space. \\

$M =$ \textquotedblleft On input $\langle x, y \rangle$, where $x$ and $y$ are binary integers greater than 0:
\begin{enumerate}
\item Let $i$, $j$, $k$, $n_x$, $n_y$, $m$ and $n$ be binary integers, and $s$, $c$, $b_i$ and $b_j$ be binary digits on work tape.
\item Let $n_x$ = $|x|$, $n_y$ = $|y|$, and $m$ be maximum of $n_x$ and $n_y$.
\item Run $C$ on input $\langle x, y \rangle$. If $C$ accepts then $n = m + 1$. Otherwise $n = m$.
\item Let $i = 1$, and $j = n$.
\item Repeat for $n$ steps:
\item \hspace*{0.5cm} Let $k = 1$, and $c = 0$.
\item \hspace*{0.5cm} Repeat for $m$ steps:
\item \hspace*{1.0cm} Let $s$ be the sum of $c$ and $k^{th}$ binary digits of $x$ and $y$. If either $x$ or $y$ does not have \\
\hspace*{1.0cm} $k^{th}$ binary digit, then assume the digit to be 0.
\item \hspace*{1.0cm} If the addition produces a carry, then $c = 1$. 
Otherwise, $c = 0$.
\item \hspace*{1.0cm} If $i = k$, then $b_i = s$.
\item \hspace*{1.0cm} If $j = k$, then $b_j = s$.
\item \hspace*{1.0cm} Increment $j$ by 1.
\item \hspace*{0.5cm} If $c = 1$ Repeat for $m$ steps:
\item \textit{Accept}.\textquotedblright
\end{enumerate}

\end{proof}
